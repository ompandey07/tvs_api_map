Security Best Practices

  • Store tokens securely (never in client-side code)
  • Use HTTPS in production
  • Implement proper logout to invalidate tokens
  • Monitor API usage and implement rate limiting

Code Examples

Python API Client

tvs_client.py 
import requests
from typing import Dict, Any, Optional

class TVSAPIClient:
    def __init__(self, base_url: str = "http://localhost:8000/api"):
        self.base_url = base_url
        self.token = None
        self.session = requests.Session()
    
    def login(self, email: str, password: str) -> Dict[str, Any]:
        """Authenticate and store token"""
        response = self.session.post(f"{self.base_url}/auth/login/", json={
            "email": email,
            "password": password
        })
        data = response.json()
        if data.get('success'):
            self.token = data['token']
            self.session.headers.update({
                'Authorization': f'Token {self.token}'
            })
        return data
    
    def create_blog(self, title: str, content: str, image_path: Optional[str] = None) -> Dict[str, Any]:
        """Create a new blog post"""
        data = {'title': title, 'content': content}
        files = {}
        if image_path:
            files['image'] = open(image_path, 'rb')
        
        response = self.session.post(f"{self.base_url}/blogs/", 
                                   data=data, files=files)
        return response.json()
    
    def get_blogs(self, search: str = "", page: int = 1) -> Dict[str, Any]:
        """Get list of blogs with optional search"""
        params = {'page': page}
        if search:
            params['search'] = search
        
        response = self.session.get(f"{self.base_url}/blogs/", params=params)
        return response.json()
    
    def import_customers(self, excel_file_path: str) -> Dict[str, Any]:
        """Import customers from Excel file"""
        with open(excel_file_path, 'rb') as file:
            files = {'excel_file': file}
            data = {'action': 'import_data'}
            response = self.session.post(f"{self.base_url}/customers/", 
                                       data=data, files=files)
        return response.json()

# Usage example
client = TVSAPIClient()
login_result = client.login("admin@tvs.com", "admin@1200")
if login_result['success']:
    print(f"✅ Login successful: {login_result['message']}")
    
    # Create a blog post
    blog_result = client.create_blog(
        title="API Integration Blog",
        content="This post was created using Python!"
    )
    print(f"📝 Blog created: {blog_result['success']}")
else:
    print("❌ Login failed")

JavaScript Client

tvsAPI.js 
class TVSAPIClient {
    constructor(baseUrl = '/api') {
        this.baseUrl = baseUrl;
        this.token = localStorage.getItem('authToken');
    }

    async request(endpoint, options = {}) {
        const url = `${this.baseUrl}${endpoint}`;
        const headers = {
            ...options.headers,
            ...(this.token && { 'Authorization': `Token ${this.token}` })
        };

        const response = await fetch(url, {
            ...options,
            headers
        });

        const data = await response.json();
        
        if (!response.ok) {
            throw new Error(data.error || 'API request failed');
        }
        
        return data;
    }

    async login(email, password) {
        const data = await this.request('/auth/login/', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ email, password })
        });
        
        if (data.success) {
            this.token = data.token;
            localStorage.setItem('authToken', data.token);
        }
        
        return data;
    }

    async createBlog(title, content, image = null) {
        const formData = new FormData();
        formData.append('title', title);
        formData.append('content', content);
        if (image) formData.append('image', image);

        return await this.request('/blogs/', {
            method: 'POST',
            body: formData
        });
    }

    async getBlogs(search = '', page = 1) {
        const params = new URLSearchParams();
        if (search) params.append('search', search);
        params.append('page', page);

        return await this.request(`/blogs/?${params}`);
    }

    async importCustomers(excelFile) {
        const formData = new FormData();
        formData.append('action', 'import_data');
        formData.append('excel_file', excelFile);

        return await this.request('/customers/', {
            method: 'POST',
            body: formData
        });
    }
}

// Usage example
const api = new TVSAPIClient();

// Login
api.login('admin@tvs.com', 'admin@1200')
    .then(result => {
        console.log('✅ Login:', result.message);
        
        // Create blog
        return api.createBlog('JS Blog', 'Created with JavaScript!');
    })
    .then(result => {
        console.log('📝 Blog created:', result.success);
    })
    .catch(error => {
        console.error('❌ Error:', error.message);
    });

Basic Test Example

test_api.py 
from django.test import TestCase
from rest_framework.test import APIClient
from rest_framework import status
from django.contrib.auth.models import User

class TVSAPITestCase(TestCase):
    def setUp(self):
        self.client = APIClient()
        self.user = User.objects.create_user(
            username='testuser',
            email='test@tvs.com',
            password='testpass123'
        )
        
    def test_authentication(self):
        response = self.client.post('/api/auth/login/', {
            'email': 'test@tvs.com',
            'password': 'testpass123'
        })
        self.assertEqual(response.status_code, status.HTTP_200_OK)
        self.assertTrue(response.json()['success'])
        self.assertIn('token', response.json())
    
    def test_blog_creation(self):
        # Login first
        login_response = self.client.post('/api/auth/login/', {
            'email': 'test@tvs.com',
            'password': 'testpass123'
        })
        token = login_response.json()['token']
        
        # Create blog
        self.client.credentials(HTTP_AUTHORIZATION=f'Token {token}')
        response = self.client.post('/api/blogs/', {
            'title': 'Test Blog',
            'content': 'Test content'
        })
        self.assertEqual(response.status_code, status.HTTP_201_CREATED)
        self.assertTrue(response.json()['success'])

    postgresql-client \

# Collect static files
